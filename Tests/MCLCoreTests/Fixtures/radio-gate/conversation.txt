# Skripty rozhovorového ramene brány P6b (Tools/radio-ref-gen/RigTranscriptGen.java,
# Tests/MCLCoreTests/Radio/RigConversationParityTests.swift). Oba je čtou z tohoto souboru; každý řádek
# scénáře je vstupem položky (otisk), změna = „PŘEGENERUJ REFERENCI".
#
# Řádek = slovo TAB pole TAB pole…; jen ASCII, žádné bílé znaky na konci řádku. Escapy v polích:
# \n \r \t \\, \xNN = znak U+00NN (v odpovědích serveru = bajt NN), \uXXXX = UTF-16 jednotka (jen texty volání).
#   section <jméno>                 položka reference (rigctld, cat-cw, rig-poller, rotctld, fldigi, n1mm-udp, otrsp, winkeyer)
#   default <klíč> <odpověď>…        výchozí odpovědi sekce (scénář je přebije stejným klíčem)
#   fallback <odpověď>              odpověď na příkaz bez klíče
#   scenario <jméno> [volba=hodnota]…   timeout=<ms javového klienta>, xvtr=<ifLo>:<ifHi>:<offset>:<0|1>[,…],
#                                   port=free (nikdo neposlouchá), rig=stub (RigPoller nad náhradou)
#   ans <klíč> <odpověď 1> <odpověď 2>…   odpověď na k-tý výskyt klíče (poslední se opakuje); klíč = celý řádek
#                                   příkazu, jinak první slovo. Odpověď: bajty; <silent> = mlčet; <delay> na začátku
#                                   = 300 ms pauza; <close> na konci = po zápisu zavřít spojení.
#   bg <příkaz> <volání>…           až server přijme <příkaz>, druhé vlákno provede <volání> (souběh s dalším call)
#   call <volání> <argumenty>…      volání API; výsledek = řádek reference
#   stub state|fail …               stav / chyba náhrady rigu (rig=stub)
#   resp <stav|close> <tělo>        odpověď fake HTTP na k-tý požadavek (poslední se opakuje)
#   in <hex>                        bajty, které Winkeyer pošle (pak konec proudu)
#   end                             konec scénáře
# <empty> = prázdné pole (prázdný text, prázdný proud).
# Timeout: javový klient v generátoru dostane timeout=…; swiftový test používá vždy velkorysých 5 000 ms —
# mlčení vyprší v obou (spodní mez), úspěšné čtení nemá těsnou mez. Výsledky na délce timeoutu nezávisí.
