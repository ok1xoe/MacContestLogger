# Skripty rozhovorového ramene brány P7b (Tools/net-ref-gen/NetConversationGen.java,
# Tests/MCLCoreTests/Net/NetConversationParityTests.swift). Oba je čtou z tohoto souboru; každý řádek scénáře
# je vstupem položky (otisk), změna = „PŘEGENERUJ REFERENCI".
#
# Řádek = slovo TAB pole TAB pole…; jen ASCII, žádné bílé znaky na okrajích řádku. Escapy v polích:
# \n \r \t \\, \xNN = znak U+00NN (bajt NN tam, kde jde o bajty), \uXXXX = UTF-16 jednotka (jen texty volání).
# <empty> = prázdné pole. <base> v textu = http://127.0.0.1:<port náhrady HTTP>.
# Náhrady běží v tomtéž procesu jen na 127.0.0.1 s porty od OS (nikdy 4532/4533), žádné skutečné služby.
# Skript řídí jedno vlákno (strana náhrady i volání klienta) — pořadí je dané, čekání mají jen velkorysou
# pojistku (20 s), po níž se zapíše <timeout> a scénář skončí (VADA, nikdy pomalý stroj).
#
# section telnet (DxClusterClient proti řádkovému serveru)
#   scenario <jméno> [port=free] [host=<host>]
#   connect                 klient se připojí (výsledek OK / EXC)
#   accept                  náhrada přijme spojení
#   srv <bajty>             náhrada pošle bajty;  srvrep <bajty> <n> = n× totéž
#   recv                    náhrada čte do CRLF (výsledek = přijaté bajty)
#   await <n>               čekat, až klient doručí n událostí (řádky + chyby); výsledek = počet
#   send <text>             DxClusterClient.send
#   connected               isConnected()
#   eof | rst | srvclose    náhrada: shutdownOutput (FIN) | RST (SO_LINGER 0) | zavřít
#   close                   DxClusterClient.close
#
# section udp-rx (WsjtxListener, AdifUdpListener, N1mmListener; odesílač náhrady na 127.0.0.1)
#   listen <wsjtx|adif|n1mm> <host> <0|taken|same>   taken = port drží náhrada, same = port posluchače 0
#   opts <i>                volby soketu posluchače i
#   dgram <asm>             datagram na port posluchače 0;  await <n>
#   reply <asm>             WsjtxListener.send na adresu posledního Decode; náhrada přijme (bajty, zdrojový port)
#   Datagram <asm> = tokeny po mezeře: i32:<n> i64:<n> u8:<n> f64:<číslo> qs:<text> (QString UTF-8)
#   qnull (QString null) qsh:<hex> (QString se surovými bajty) hex:<hex> txt:<text UTF-8>; mezera v textu \x20.
#
# section udp-mcast (jako udp-rx; multicast posluchači se přihlásí do skupiny jen na lo0 — Swift přes testovací
#   index rozhraní, Java přesunem členství na lo0; runner, který na lo0 multicast neumí, položku přeskočí)
#
# section udp-tx (UdpBroadcaster; přijímače náhrady R0, R1 na 127.0.0.1, R6 na [::1])
#   opts | send <text> <cíl>… (R0, R1, R6 = přijímač; jinak host:port doslova) | get <R> | close
#
# section http (HamQthClient, QrzClient, ClubLogClient, ScorePoster proti náhradě HTTP/1.1)
#   scenario <jméno> [port=free]
#   resp <stav|close> <Content-Type> <tělo> [Location]   odpověď na k-tý požadavek (poslední se opakuje)
#   client <hamqth|qrz> <jméno> <heslo>;  lookup <volačka>
#   clublog <email> <heslo> <volačka> <api> <adif>;  score <cesta> <xml>
#   Callbooky jdou na https://www.hamqth.com / https://xmldata.qrz.com přepsané na <base>.
#
# section mqtt (MqttSyncTransport proti skriptovanému brokeru; strana brokeru ruční)
#   scenario <jméno> [user=<u>] [pass=<p>] [offline=<json StationStatusWire>]
#   bg <volání> [json] / join   volání na pozadí (connect, close, insert, update, delete, status, spot, msg,
#                           serial) / jeho výsledek;  call <volání> [json] = hned
#   accept | expect <TYP> (paket klienta, bajty) | expectset <n> (n paketů v pořadí nezávislém na vláknech
#   Paho, seřazené) | connack <sp> <rc> | suback <kódy hex> (na poslední SUBSCRIBE) | puback [rc] (na poslední PUBLISH)
#   publish <qos> <retain> <téma> <payload> (id brokeru 1, 2, …) | disconnect <rc> | drop (FIN)
#   connected | waitoffline (dokud isConnected) | await <n> (události posluchačů)
#   gate | release           posluchači čekají na uvolnění | uvolnit;  pending = bajty od klienta právě teď
